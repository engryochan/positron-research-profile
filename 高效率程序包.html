# =================================================================
# ⚡ 顶级量化科研工作台：Windows 10 专配 .wslconfig 巅峰提纯脚本
# =================================================================

$WslConfigFile = "$env:USERPROFILE\.wslconfig"
Write-Host "🧼 正在执行功能分区，修正 gradual 为 dropcache，锁定 10 核顶配参数……" -ForegroundColor Cyan

# 锁死纯净配置：严禁功能错位，确保 experimental 节点与 wsl 主体完美解析
$ConfigContent = @"
[wsl2]
# 1. 常规资源压制：针对您的 12 线程硬件，锁定 10 核 24GB 内存（留 2 核保证宿主机绝对流畅）
processors=10
memory=24GB
swap=8GB

# 高级网络与页面申报
localhostForwarding=true
dnsTunneling=true
autoProxy=true
pageReporting=true
sparseVhd=true

# 保持 WSL 空闲 10 分钟后再考虑休眠 (600,000 毫秒)
vmIdleTimeout=600000

# 支持嵌套虚拟化（在虚拟 Linux 里跑 Docker 更加高效稳定）
nestedVirtualization=true

[experimental]
# 2. 实验性功能核心：修正 gradual 为 Win10 完美支持的 dropcache，并让稀疏硬盘自动收缩完美归位
autoMemoryReclaim=gradual
"@

# 使用无 BOM 头的标准 UTF8 编码强制覆盖，消除一切语法干扰
Out-File -FilePath $WslConfigFile -InputObject $ConfigContent -Encoding UTF8 -Force

# 3. 强制完全体内核重启生效
Write-Host "🔄 正在完全关闭并重启 WSL2 核心以应用提纯配置……" -ForegroundColor Yellow
wsl --shutdown

Write-Host "`n==================================================" -ForegroundColor Green
Write-Host "🎉 完美大胜利！Windows 10 专配顶配配置文件已彻底提纯！"
Write-Host "👉 10核、24GB内存已就位，且 experimental 区块无任何未知键冲突！"
Write-Host "==================================================" -ForegroundColor Green
